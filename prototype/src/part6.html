/* ---------------- the compositor ---------------- drawFrame(ctx, t) paints the whole video at time t. Preview and export both call this, so what you scrub is exactly what you get. ------------------------------------------------- */ var L = null; // cached layout function drawFrame(ctx, t){ var o = L.out, ct = cardTimes(); ctx.save(); ctx.fillStyle = S.bg; ctx.fillRect(0, 0, o.w, o.h); var pageShown = 0, bodyProgress = 0; if (ct.intro && t < ct.intro){ drawCard(ctx, o, S.title, S.subtitle, fadeAlpha(t, ct.intro)); } else if (ct.outro && t >= ct.intro + ct.body){ drawCard(ctx, o, S.outro, "", fadeAlpha(t - ct.intro - ct.body, ct.outro)); } else { var bt = Math.min(ct.body, Math.max(0, t - ct.intro)); bodyProgress = ct.body > 0 ? bt / ct.body : 0; if (S.mode === "slide" || S.mode === "kenburns"){ pageShown = drawSlides(ctx, o, bodyProgress); } else { pageShown = drawScroll(ctx, o, bodyProgress); } drawOverlays(ctx, o, pageShown, bodyProgress); } ctx.restore(); } function fadeAlpha(t, span){ var f = Math.min(0.45, span * 0.28); if (t < f) return t / f; if (t > span - f) return Math.max(0, (span - t) / f); return 1; } /* --- continuous / auto pace --- */ function drawScroll(ctx, o, prog){ var travel = Math.max(0, L.total - o.h); var p = prog; if (S.mode === "autopace"){ p = autopaceCurve(prog); } else if (S.ease){ p = easeInOut(prog); } // A single page shorter than the frame has nowhere to travel: centre it. var camY = travel > 0 ? p * travel : -(o.h - L.total)/2; var shown = 1; for (var i = 0; i < L.items.length; i++){ var it = L.items[i]; var y = it.y - camY; if (y > o.h || y + it.h < 0) continue; if (y <= o.h * 0.45 && y + it.h >= o.h * 0.45) shown = it.page.num; drawPage(ctx, it.page, it.x, y, it.w, it.h); } return shown; } // Maps linear time onto distance so text-heavy pages scroll more slowly. function autopaceCurve(prog){ var w = pageWeights(), acc = 0, i; var tAcc = [0], dAcc = [0], distTotal = L.total, d = 0; for (i = 0; i < L.items.length; i++){ acc += w[i]; d += L.items[i].h + (i < L.items.length-1 ? (L.items[1] ? L.items[1].y - L.items[0].y - L.items[0].h : 0) : 0); tAcc.push(acc); dAcc.push(Math.min(1, d / distTotal)); } for (i = 1; i < tAcc.length; i++){ if (prog <= tAcc[i]){ var span = tAcc[i] - tAcc[i-1] || 1; var local = (prog - tAcc[i-1]) / span; return dAcc[i-1] + (dAcc[i] - dAcc[i-1]) * local; } } return 1; } /* --- slide / ken burns --- */ function drawSlides(ctx, o, prog){ var w = pageWeights(), n = L.items.length; var acc = 0, idx = 0, local = 0; for (var i = 0; i < n; i++){ if (prog <= acc + w[i] || i === n-1){ idx = i; local = w[i] ? (prog - acc)/w[i] : 0; break; } acc += w[i]; } local = Math.min(1, Math.max(0, local)); var tr = S.transition, trDur = 0.22; // share of a page's time spent transitioning var cur = L.items[idx], nxt = L.items[idx+1]; if (S.mode === "kenburns"){ drawKB(ctx, o, cur, local, 1); if (nxt && local > 1-trDur && tr !== "cut"){ drawKB(ctx, o, nxt, 0, (local - (1-trDur))/trDur); } return cur.page.num; } var inTr = nxt && local > 1 - trDur && tr !== "cut"; var k = inTr ? (local - (1-trDur)) / trDur : 0; k = easeInOut(Math.min(1, k)); if (!inTr){ drawFitted(ctx, o, cur, 0, 0, 1); } else if (tr === "fade"){ drawFitted(ctx, o, cur, 0, 0, 1 - k); drawFitted(ctx, o, nxt, 0, 0, k); } else if (tr === "push"){ drawFitted(ctx, o, cur, 0, -o.h * k, 1); drawFitted(ctx, o, nxt, 0, o.h * (1-k), 1); } else { drawFitted(ctx, o, cur, -o.w * k, 0, 1); drawFitted(ctx, o, nxt, o.w * (1-k), 0, 1); } return (inTr && k > .5 ? nxt : cur).page.num; } function drawFitted(ctx, o, it, dx, dy, alpha){ var pad = L.pad; var sc = Math.min((o.w - pad*2) / it.page.w, (o.h - pad*2) / it.page.h); if (S.fit === "width") sc = (o.w - pad*2) / it.page.w; var w = it.page.w * sc, h = it.page.h * sc; var x = (o.w - w)/2 + dx; var y = (o.h - h)/2 + dy; if (S.fit === "width" && h > o.h - pad*2) y = pad + dy; // tall page: pin to top ctx.save(); ctx.globalAlpha = alpha; drawPage(ctx, it.page, x, y, w, h); ctx.restore(); } function drawKB(ctx, o, it, local, alpha){ var zoom = 1.06 + 0.07 * local; var pad = L.pad; var sc = Math.min((o.w - pad*2)/it.page.w, (o.h - pad*2)/it.page.h) * zoom; var w = it.page.w * sc, h = it.page.h * sc; var drift = (h > o.h) ? (h - o.h) * local : 0; ctx.save(); ctx.globalAlpha = alpha; drawPage(ctx, it.page, (o.w - w)/2, (o.h - h)/2 - (h > o.h ? (h-o.h)/2 - drift : 0), w, h); ctx.restore(); } function drawPage(ctx, page, x, y, w, h){ ctx.save(); ctx.shadowColor = "rgba(0,0,0,.45)"; ctx.shadowBlur = Math.max(8, w * 0.012); ctx.shadowOffsetY = Math.max(2, w * 0.003); ctx.fillStyle = "#fff"; ctx.fillRect(x, y, w, h); ctx.restore(); ctx.drawImage(page.canvas, x, y, w, h); }
